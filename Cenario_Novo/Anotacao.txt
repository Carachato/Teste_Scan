function gravarArquivoTxt(NmServidor, HrJs, PaginaJs, TrasacaoJs, TrancodeJs) {
    try {
        var fso = Server.CreateObject("Scripting.FileSystemObject");
        var Caminho = "C:\\inetpub\\wwwroot\\_ProjetoIIS\\Rastreamento\\"
		var Extencao = ".txt"
        var caminhoCompleto = Caminho + NmServidor + " - " + HrJs + " - " + PaginaJs + " - " + TrasacaoJs + "." + Extencao;
        var file = fso.CreateTextFile(caminhoCompleto, true);
        file.WriteLine(TrancodeJs);
        file.Close();        
        return true;
    } catch(e) {
        Response.Write("Erro interno no script JS: " + Caminho + e.description);
        return false;
    }
}
